const double PRECIO_HORA = 1.50;
const double PRECIO_DIA  = 5.00;

class Tarifa {                         // la interfaz
public:
    virtual double cobrar(int horas) const = 0;
    virtual ~Tarifa() = default;
};

class TarifaPorHora : public Tarifa {  // estrategia 1
public:
    double cobrar(int horas) const override {
        return PRECIO_HORA * horas;
    }
};

      
class TarifaFija : public Tarifa {     // estrategia 2
public:
    double cobrar(int) const override {
        return PRECIO_DIA;
    }
};

class Estacionamiento {                // el contexto
    const Tarifa* tarifa;
public:
    Estacionamiento(const Tarifa& t) : tarifa(&t) {}
    void cambiarTarifa(const Tarifa& t) { tarifa = &t; }
    void salida(int horas) const {
        cout << horas << " h: $"
             << tarifa->cobrar(horas) << endl;
    }
};

      
int main() {
    cout << fixed << setprecision(2);
    TarifaPorHora porHora;
    TarifaFija fija;
    Estacionamiento e(porHora);
    e.salida(2);
    e.salida(6);
    e.cambiarTarifa(fija);
    e.salida(2);
    e.salida(6);
    return 0;
}

class Observador {
public:
    virtual void avisar(int cuenta, double saldo) = 0;
    virtual ~Observador() = default;
};
El plano de cualquier observador: una sola función, avisar.

      
class AvisoSMS : public Observador {
public:
    void avisar(int cuenta, double saldo) override {
        cout << "SMS: la cuenta " << cuenta
             << " tiene $" << saldo << endl;
    }
};

class Auditoria : public Observador {
    int movimientos = 0;
public:
    void avisar(int cuenta, double) override {
        movimientos++;
        cout << "Auditoria: movimiento " << movimientos
             << " en " << cuenta << endl;
    }
};

      
class Cuenta {
    int numero;
    double saldo = 0;
    Observador* obs[5];
    int n = 0;
public:
    Cuenta(int num) : numero(num) {}
    void suscribir(Observador* o) {
        if (n < 5) obs[n++] = o;
    }
    void depositar(double monto) {
        saldo += monto;
        for (int i = 0; i < n; i++)
            obs[i]->avisar(numero, saldo);
    }
};

      
int main() {
    cout << fixed << setprecision(2);
    AvisoSMS sms;
    Auditoria audit;
    Cuenta c(1001);
    c.depositar(50);          // nadie escucha todavia
    c.suscribir(&sms);
    c.depositar(100);
    c.suscribir(&audit);
    c.depositar(25);
    return 0;
}

Cuatro mensajes de g++

  

    

    

      
Tarifa t;
error: cannot declare variable 't' to be of abstract type 'Tarifa'
note:   because the following virtual functions are pure within 'Tarifa':
note:     'virtual double Tarifa::cobrar(int) const'

      
Una clase con una función = 0 es abstracta: es un plano, no una casa.
Se crean objetos de las clases hijas (TarifaFija) y se usan con una referencia o un puntero a Tarifa.

      
double cobra(int) const override {
    return PRECIO_DIA;
}
error: 'double TarifaFija::cobra(int) const' marked 'override', but does not override

      
El nombre es cobrar, no cobra.
override le pide a g++ que confirme que la función de verdad reemplaza una del padre. Sin él, este error compila y la tarifa nunca se usa. Lo mismo pasa si falta el const.

      
Config c = Config::instancia();
error: use of deleted function 'Config::Config(const Config&)'

      
Una copia sería un segundo objeto: el Singleton lo prohíbe.
Se guarda una referencia: Config& c = Config::instancia();

      
Tarifa* t = new TarifaFija;
cout << t->cobrar(3) << endl;
delete t;
warning: deleting object of abstract class type 'Tarifa' which has non-virtual destructor will cause undefined behavior [-Wdelete-non-virtual-dtor]

      
Compila, pero borrar por el puntero del padre no llama al destructor del hijo.
Toda interfaz lleva virtual ~Tarifa() = default;, como en el ejemplo de Strategy.

Lo que se lleva de esta unidad

  

    
Patrones
Planos probados con nombre. Singleton con cuidado; Strategy y Observer separan lo que cambia.

    
SOLID
Una razón para cambiar; extender sin modificar; hijos sustituibles; interfaces pequeñas; depender de abstracciones.

    
UML
Tres compartimientos, + - #. Triángulo: herencia. Rombo lleno: composición. Rombo vacío: agregación.

    
Refactorizar
Renombrar, nombrar constantes, extraer funciones. Pasos pequeños y la salida comparada en cada uno.

    
Código limpio
DRY: un lugar por regla. KISS: lo simple. YAGNI: solo lo que se pidió.

    
Diseño
Alta cohesión, bajo acoplamiento, responsabilidades separadas y composición antes que herencia.
